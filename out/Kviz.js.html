Kviz.js

import Kerdes from "./Kerdes.js";

/**
 * @class Kviz
 * @classdesc Egy kvíz vezérléséért felelős osztály.
 * Kezeli a kérdések megjelenítését, pontszámot és a folyamatot.
 */
export default class Kviz {
  /**
   * @private
   * @type {Array<Object>}
   * @description A kérdések listája.
   */
  #kerdesek = [];

  /**
   * @private
   * @type {number}
   * @description Az aktuális kérdés indexe.
   */
  #aktualisIndex;

  /**
   * @private
   * @type {number}
   * @description A játékos pontszáma.
   */
  #pontszam;

  /**
   * @constructor
   * @param {Array<Object>} kerdesAdatok - A kérdések adatai.
   * @param {HTMLElement} szuloElem - A megjelenítés cél DOM elem.
   */
  constructor(kerdesAdatok = [], szuloElem) {
    this.#kerdesek = kerdesAdatok;
    this.#aktualisIndex = 0;
    this.#pontszam = 0;
    this.szuloElem = szuloElem;
    this.#kever(this.#kerdesek);

    this.#ujKerdes();
  }

  /**
   * @private
   * @method kever
   * @param {Array} tomb - A kérdések tömbje.
   * @returns {void}
   * @description A kérdések sorrendjének összekeverése.
   */
  #kever(tomb) {
    for (let i = tomb.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [tomb[i], tomb[j]] = [tomb[j], tomb[i]];
    }
  }

  /**
   * @private
   * @method ujKerdes
   * @returns {void}
   * @description Új kérdés megjelenítése vagy az eredmény kiírása.
   */
  #ujKerdes() {
    this.szuloElem.innerHTML = "";

    if (this.vegeVanE()) {
      const eredmeny = document.createElement("h3");
      eredmeny.textContent = `Eredmény: ${this.#pontszam} / ${this.#kerdesek.length}`;
      this.szuloElem.appendChild(eredmeny);

      const resetGomb = document.createElement("button");
      resetGomb.textContent = "Újraindítás";
      resetGomb.classList.add("reset-gomb");

      resetGomb.addEventListener("click", () => {
        const modal = document.getElementById("modal");
        const igen = document.getElementById("igen");
        const nem = document.getElementById("nem");

        modal.classList.remove("hidden");

        igen.onclick = () => {
          modal.classList.add("hidden");
          this.reset();
        };

        nem.onclick = () => {
          modal.classList.add("hidden");
        };
      });

      this.szuloElem.appendChild(resetGomb);

      return;
    }

    const info = document.createElement("p");
    info.classList.add("melyik");
    info.textContent = `⚡ ${this.#aktualisIndex + 1} / ${this.#kerdesek.length} kérdés ⚡`;
    this.szuloElem.appendChild(info);

    const adat = this.#kerdesek[this.#aktualisIndex];
    this.aktualisKerdesObj = new Kerdes(
      adat.kerdesSzoveg,
      adat.valaszok,
      this.szuloElem,
      this,
    );

    const resetGomb = document.createElement("button");
    resetGomb.textContent = "Újraindítás";
    resetGomb.classList.add("reset-gomb");

    resetGomb.addEventListener("click", () => {
      const modal = document.getElementById("modal");
      const igen = document.getElementById("igen");
      const nem = document.getElementById("nem");

      modal.classList.remove("hidden");

      igen.onclick = () => {
        modal.classList.add("hidden");
        this.reset();
      };

      nem.onclick = () => {
        modal.classList.add("hidden");
      };
    });

    this.szuloElem.appendChild(resetGomb);
  }

  /**
   * @method reset
   * @returns {void}
   * @description A kvíz újraindítása.
   */
  reset() {
    this.#aktualisIndex = 0;
    this.#pontszam = 0;
    this.#kever(this.#kerdesek);
    this.#ujKerdes();
  }

  /**
   * @method getPontszam
   * @returns {number}
   * @description Az aktuális pontszám visszaadása.
   */
  getPontszam() {
    return this.#pontszam;
  }

  /**
   * @method getKerdesekSzama
   * @returns {number}
   * @description A kérdések számának visszaadása.
   */
  getKerdesekSzama() {
    return this.#kerdesek.length;
  }

  /**
   * @method getAktualisKerdes
   * @returns {Object}
   * @description Az aktuális kérdés visszaadása.
   */
  getAktualisKerdes() {
    return this.#kerdesek[this.#aktualisIndex];
  }

  /**
   * @method valaszEllenorzese
   * @param {Object} valasz - A kiválasztott válasz objektum.
   * @returns {boolean}
   * @description Ellenőrzi, hogy a válasz helyes-e, és növeli a pontszámot.
   */
  valaszEllenorzese(valasz) {
    const helyes = this.aktualisKerdesObj.helyesValaszE(valasz);

    if (helyes) {
      this.#pontszam++;
    }

    return helyes;
  }

  /**
   * @method valaszKattintva
   * @param {Object} valasz - A kiválasztott válasz.
   * @returns {void}
   * @description Válasz kattintás kezelése és következő kérdés betöltése.
   */
  valaszKattintva(valasz) {
    this.valaszEllenorzese(valasz);

    setTimeout(() => {
      this.#aktualisIndex++;
      this.#ujKerdes();
    }, 100);
  }

  /**
   * @method vegeVanE
   * @returns {boolean}
   * @description Megállapítja, hogy a kvíz véget ért-e.
   */
  vegeVanE() {
    return this.#aktualisIndex >= this.#kerdesek.length;
  }
}
